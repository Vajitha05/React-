UserDetails.jsx

import React from "react";
import { useParams, useNavigate } from "react-router-dom";

function UserDetails() {
  const { id } = useParams();
  const navigate = useNavigate();

  return (
    <div>
      <h2>User Details</h2>

      <p>User ID: {id}</p>

      <button onClick={() => navigate("/users")}>
        Back to Users
      </button>
    </div>
  );
}

export default UserDetails;

Users.jsx

import React from "react";
import { Link } from "react-router-dom";
function Users() {
  const users = [
    { id: 1, name: "John" },
    { id: 2, name: "Alice" },
    { id: 3, name: "David" },
  ];

  return (
    <div>
      <h2>Users Page</h2>

      <ul>
        {users.map((user) => (
          <li key={user.id}>
            <Link to={`/users/${user.id}`}>
              {user.name}
            </Link>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default Users;
